Esquema Relacional (Pasaje a Tablas) - Base de Datos: sga_db

🗄️ sga_db
roles(id_rol, nombre)
Atributo Descripción
id_rol PK INT AUTO_INCREMENT Identificador único y correlativo del rol.
nombre VARCHAR(50) UNIQUE Nombre del rol del sistema (ej. Administrador, Estudiante, Docente).
estados_asignaturas(id_estado_asignatura, nombre)
Atributo Descripción
id_estado_asignatura PK INT AUTO_INCREMENT Identificador único del estado de la asignatura.
nombre VARCHAR(50) UNIQUE Descripción del estado (ej. Activa, Inactiva, Suspendida).
tipos_evaluaciones(id_tipo_evaluacion, nombre)
Atributo Descripción
id_tipo_evaluacion PK INT AUTO_INCREMENT Identificador único del tipo de evaluación.
nombre VARCHAR(50) UNIQUE Nombre del tipo de prueba (ej. Primer Parcial, Examen Final, Trabajo Obligatorio).
usuarios(id_usuario, cedula_identidad, primer_nombre, segundo_nombre, primer_apellido, segundo_apellido, email, contrasenia, fecha_nacimiento, id_rol, activo, fecha_registro)
Atributo Descripción
id_usuario PK INT AUTO_INCREMENT Identificador único del usuario en la superclase.
cedula_identidad VARCHAR(10) UNIQUE Documento de identidad oficial del usuario.
primer_nombre VARCHAR(50) Primer nombre obligatorio del usuario.
segundo_nombre VARCHAR(50) NULL Segundo nombre opcional del usuario.
primer_apellido VARCHAR(50) Primer apellido obligatorio del usuario.
segundo_apellido VARCHAR(50) NULL Segundo apellido opcional del usuario.
email VARCHAR(150) UNIQUE Dirección de correo electrónico única para acceso y notificaciones.
contrasenia VARCHAR(50) Clave de acceso codificada del usuario.
fecha_nacimiento DATE Fecha de nacimiento registrada.
id_rol FK INT Clave foránea referenciada a la tabla roles(id_rol).
activo TINYINT(1) DEFAULT 1 Indica si la cuenta se encuentra habilitada (1) o inactiva (0).
fecha_registro TIMESTAMP Fecha y hora automática de alta en el sistema.
estudiantes(id_usuario, nro_legajo)
Atributo Descripción
id_usuario PK FK INT Identificador heredado de usuarios(id_usuario) (Relación de subclase).
nro_legajo VARCHAR(20) UNIQUE Número de legajo estudiantil único asignado al alumno.
docentes(id_usuario, nro_legajo)
Atributo Descripción
id_usuario PK FK INT Identificador heredado de usuarios(id_usuario) (Relación de subclase).
nro_legajo VARCHAR(20) UNIQUE Número de legajo docente único asignado al profesor.
asignaturas(id_asignatura, nombre, creditos, cupo_maximo, id_estado_asignatura, activo)
Atributo Descripción
id_asignatura PK INT AUTO_INCREMENT Identificador único de la asignatura.
nombre VARCHAR(150) UNIQUE Nombre completo de la asignatura o materia.
creditos INT (>0) Cantidad de créditos curriculares que otorga la materia.
cupo_maximo INT (>=0) Capacidad máxima de alumnos permitidos para cursar.
id_estado_asignatura FK INT Referencia al estado en estados_asignaturas(id_estado_asignatura).
activo TINYINT(1) DEFAULT 1 Estado operacional de la asignatura (1=Habilitada, 0=Deshabilitada).
inscripciones(id_inscripcion, id_usuario, id_asignatura, fecha_inscripcion)
Atributo Descripción
id_inscripcion PK INT AUTO_INCREMENT Identificador correlativo único del registro de inscripción.
id_usuario FK INT Estudiante que realiza el alta (referencia a estudiantes(id_usuario)).
id_asignatura FK INT Asignatura elegida (referencia a asignaturas(id_asignatura)).
fecha_inscripcion DATE Fecha de registro formal de la inscripción.
calificaciones(id_calificacion, id_inscripcion, id_usuario, id_tipo_evaluacion, nota, fecha_calificacion)
Atributo Descripción
id_calificacion PK INT AUTO_INCREMENT Identificador único del registro de nota.
id_inscripcion FK INT Inscripción vinculada (referencia a inscripciones(id_inscripcion)).
id_usuario FK INT Docente evaluador que asigna la nota (referencia a docentes(id_usuario)).
id_tipo_evaluacion FK INT Instancia evaluada (referencia a tipos_evaluaciones(id_tipo_evaluacion)).
nota DECIMAL(4,2) (1..12) Calificación numérica obtenida (rango legal entre 1.00 y 12.00).
fecha_calificacion DATE Fecha de ingreso de la nota al sistema.
asignaturas_docentes(id_usuario, id_asignatura, fecha_asignacion)
Atributo Descripción
id_usuario PK FK INT Docente asignado (referencia a docentes(id_usuario)). Parte de PK compuesta.
id_asignatura PK FK INT Materia dictada (referencia a asignaturas(id_asignatura)). Parte de PK compuesta.
fecha_asignacion DATE Fecha de designación del docente en la asignatura.
